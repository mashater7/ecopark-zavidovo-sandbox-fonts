<script>
(function(){
  /* ── переключатели ─────────────────────────────────────────────────── */
  var IMYA        = 'Другие новости';
  var VSE_NOVOSTI = '/news';
  var SKOLKO      = 3;
  var KROME_ETOY  = true;
  var DATY_RU     = true;
  var UBRAT_EMOJI = true;

  var MES = {january:'января',february:'февраля',march:'марта',april:'апреля',
    may:'мая',june:'июня',july:'июля',august:'августа',september:'сентября',
    october:'октября',november:'ноября',december:'декабря',
    jan:'января',feb:'февраля',mar:'марта',apr:'апреля',jun:'июня',jul:'июля',
    aug:'августа',sep:'сентября',sept:'сентября',oct:'октября',nov:'ноября',dec:'декабря'};
  var EMO = /[\u{1F300}-\u{1FAFF}\u{2600}-\u{27BF}\u{FE0F}\u{200D}]/gu;

  function poRusski(t){
    var s = t.toLowerCase(), mon = null, dlina = 0;
    for (var k in MES) if (s.indexOf(k) > -1 && k.length > dlina) { mon = k; dlina = k.length; }
    if (!mon) return null;
    var bezGoda = s.replace(/20\d{2}/,'');
    var d = bezGoda.match(/\d{1,2}/);
    if (!d) return null;
    var y = s.match(/20\d{2}/);
    return parseInt(d[0],10) + ' ' + MES[mon] + (y ? ' ' + y[0] : '');
  }

  function chistyPut(h){
    if (!h) return '';
    return h.replace(/^https?:\/\/[^\/]+/,'').replace(/[?#].*$/,'').replace(/\/+$/,'').toLowerCase();
  }

  function snyatVysoty(rec){
    [].forEach.call(rec.querySelectorAll('.t404__title,.t404__descr,.t404__col'), function(el){
      if (el.style.height && el.style.height !== 'auto') el.style.height = 'auto';
    });
  }

  function odet(rec){
    if (rec.classList.contains('azNHide')) return;            /* лента Экопарка — не наша */
    if (rec.getAttribute('data-azd') === 'y') { snyatVysoty(rec); return; }
    rec.setAttribute('data-azd','y');

    var lenta = rec.querySelector('.t404'), cont = rec.querySelector('.t-container');
    if (!lenta || !cont) return;

    /* Тильда кладёт по три карточки в СВОЙ .t-container (это ряд).
       Из-за этого скрытая новость оставляет в ряду дыру, а следующая
       карточка уезжает вниз. Поэтому нужные карточки собираются
       в первый контейнер, остальные контейнеры гасятся. */
    var conts = [].slice.call(rec.querySelectorAll('.t-container'));

    /* шапка секции */
    var h = document.createElement('div');
    h.className = 'azd-h';
    h.innerHTML = '<div class="azd-hin"><h3 class="azd-h2"></h3><a class="azd-all"></a></div>';
    h.querySelector('.azd-h2').textContent = IMYA;
    var a = h.querySelector('.azd-all');
    a.textContent = 'Все новости';
    a.setAttribute('href', VSE_NOVOSTI);
    lenta.insertBefore(h, cont);

    /* отбор карточек: без текущей новости, не больше SKOLKO.
       `?azput=/vivayachts` — подмена адреса, чтобы проверять отбор
       в локальном превью (на боевой параметра нет, берётся pathname). */
    var podmena = null;
    try { podmena = new URLSearchParams(location.search).get('azput'); } catch (e) {}
    var tut = chistyPut(podmena || location.pathname);
    var ostalos = 0;
    [].forEach.call(rec.querySelectorAll('.t404__col'), function(col){
      var ssylka = col.querySelector('a');
      var put = chistyPut(ssylka ? ssylka.getAttribute('href') : '');
      var eto_tekushchaya = KROME_ETOY && put && put === tut;
      if (eto_tekushchaya || (SKOLKO > 0 && ostalos >= SKOLKO)) { col.classList.add('azd-skip'); return; }
      col.style.setProperty('--d', ostalos);
      ostalos++;
      if (col.parentNode !== cont) cont.appendChild(col);   /* все выбранные — в один ряд */

      var dt = col.querySelector('.t404__date');
      if (DATY_RU && dt) { var r = poRusski(dt.textContent.trim()); if (r) dt.textContent = r; }
      if (UBRAT_EMOJI) {
        [].forEach.call(col.querySelectorAll('.t404__title,.t404__descr'), function(el){
          el.textContent = el.textContent.replace(EMO,'').replace(/\s{2,}/g,' ').trim();
        });
      }
    });

    /* если показывать нечего — секции на странице не будет */
    if (!ostalos) { rec.style.display = 'none'; return; }

    /* лишние ряды-контейнеры гасим, чтобы под рядом не осталось пустот */
    conts.forEach(function(c, i){ if (i > 0) c.classList.add('azd-row-off'); });
    snyatVysoty(rec);

    var zhivye = [].filter.call(rec.querySelectorAll('.t404__col'), function(c){
      return !c.classList.contains('azd-skip'); });

    function pokazat(el){ el.classList.add('azd-in'); }
    if ('IntersectionObserver' in window) {
      var io = new IntersectionObserver(function(es){
        es.forEach(function(e){ if (e.isIntersecting) { pokazat(e.target); io.unobserve(e.target); } });
      }, {rootMargin:'0px 0px -8% 0px', threshold:.05});
      zhivye.forEach(function(c){ io.observe(c); });
      setTimeout(function(){ zhivye.forEach(pokazat); }, 2200);
    } else {
      zhivye.forEach(pokazat);
    }

    [400, 1200, 2500].forEach(function(ms){ setTimeout(function(){ snyatVysoty(rec); }, ms); });
  }

  function start(){
    var demo = document.getElementById('azd-demo');
    if (demo && document.getElementById('allrecords')) demo.remove();
    [].forEach.call(document.querySelectorAll('.r[data-record-type="404"],.r[data-alias-record-type="404"]'), odet);
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', start);
  else start();
  window.addEventListener('load', start);
})();
</script>
