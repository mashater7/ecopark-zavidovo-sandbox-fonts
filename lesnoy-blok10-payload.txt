<script>
(function(){
  var root=document.getElementById('lesAround'); if(!root) return;
  var caps={1:'Футбольное поле, площадки и спорт рядом с домом',
            2:'Река Дойбица по границе посёлка, лес начинается в ста метрах',
            3:'Освещённые улицы, закрытый периметр, круглосуточная охрана',
            4:'Свой выезд на М-11, станция Путепроводная в двух километрах'};
  function act(row){
    root.querySelectorAll('.la-row').forEach(function(r){ r.classList.toggle('on', r===row); });
    var k=row.getAttribute('data-k');
    root.querySelectorAll('.la-stage img').forEach(function(im){ im.classList.toggle('on', im.getAttribute('data-k')===k); });
    root.querySelector('.la-cap').textContent=caps[k];
  }
  root.querySelectorAll('.la-row').forEach(function(row){
    row.addEventListener('mouseenter', function(){ act(row); });
    row.addEventListener('click', function(){ act(row); });
    row.addEventListener('focus', function(){ act(row); });
  });
})();
</script>
