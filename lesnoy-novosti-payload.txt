<script>
(function(){
  /* ── переключатели ─────────────────────────────────────────────────── */
  var ZAGOLOVOK   = true;
  var IMYA        = 'Новости посёлка';
  var VSE_NOVOSTI = '/news';
  var DATY_RU     = true;
  var UBRAT_EMOJI = true;
  /* Кодовый блок «Жизнь посёлка» (шесть демо-карточек со ссылками на
     akvatoriya-news-item.html — такой страницы на kzavidovo.ru нет).
     Лента заменяет его. false — вернуть блок на страницу. */
  var SPRYATAT_KODOVYY = false;
  var KODOVYY = '';

  var MES = {january:'января',february:'февраля',march:'марта',april:'апреля',
    may:'мая',june:'июня',july:'июля',august:'августа',september:'сентября',
    october:'октября',november:'ноября',december:'декабря',
    jan:'января',feb:'февраля',mar:'марта',apr:'апреля',jun:'июня',jul:'июля',
    aug:'августа',sep:'сентября',sept:'сентября',oct:'октября',nov:'ноября',dec:'декабря'};
  var SKOLKO = 12;   /* сколько новостей оставить в карусели */
  var EMO = /[\u{1F300}-\u{1FAFF}\u{2600}-\u{27BF}\u{FE0F}\u{200D}]/gu;

  function poRusski(t){
    var s = t.toLowerCase(), mon = null;
    var dlinnye = ['january','february','march','april','september','october','november','december','june','july','august','may'];
    for (var i = 0; i < dlinnye.length; i++) if (s.indexOf(dlinnye[i]) > -1) { mon = dlinnye[i]; break; }
    if (!mon) { var m = s.match(/\b(jan|feb|mar|apr|may|jun|jul|aug|sept|sep|oct|nov|dec)\b/); if (m) mon = m[1]; }
    if (!mon) return null;                       /* дата уже не английская */
    var bezGoda = s.replace(/20\d{2}/,'');
    var d = bezGoda.match(/\d{1,2}/);
    if (!d) return null;
    var y = s.match(/20\d{2}/);
    return parseInt(d[0],10) + ' ' + MES[mon] + (y ? ' ' + y[0] : '');
  }

  function snyatVysoty(rec){
    [].forEach.call(rec.querySelectorAll('.t404__title,.t404__descr,.t404__col'), function(el){
      if (el.style.height && el.style.height !== 'auto') el.style.height = 'auto';
    });
  }

  function odet(rec){
    if (rec.getAttribute('data-azl') === 'y') { snyatVysoty(rec); return; }
    rec.setAttribute('data-azl','y');

    var lenta = rec.querySelector('.t404'), cont = rec.querySelector('.t-container');
    if (!lenta || !cont) return;

    /* Тильда раскладывает ленту по рядам: у каждого ряда свой .t-container.
       Собираем все карточки в первый контейнер, лишние ряды убираем,
       иначе горизонтальной прокрутке нечего листать. */
    (function(){
      var ryady = lenta.querySelectorAll('.t-container');
      if (ryady.length < 2) return;
      for (var i = 1; i < ryady.length; i++) {
        var kartochki = ryady[i].querySelectorAll('.t404__col');
        for (var j = 0; j < kartochki.length; j++) cont.appendChild(kartochki[j]);
        ryady[i].parentNode.removeChild(ryady[i]);
      }
    })();

    /* оставляем только свежие */
    (function(){
      var vse = cont.querySelectorAll('.t404__col');
      for (var i = SKOLKO; i < vse.length; i++) vse[i].parentNode.removeChild(vse[i]);
    })();

    /* шапка секции */
    if (ZAGOLOVOK) {
      var h = document.createElement('div');
      h.className = 'azl-h';
      h.innerHTML = '<h2 class="azl-h2"></h2><a class="azl-all"></a>';
      h.querySelector('.azl-h2').textContent = IMYA;
      var a = h.querySelector('.azl-all');
      a.textContent = 'Все новости';
      a.setAttribute('href', VSE_NOVOSTI);
      lenta.insertBefore(h, cont);
    } else {
      rec.classList.add('azl-noh');
    }

    /* прокрутка ленты стрелками по краям */
    (function(){
      var knopki = [];
      ['-1','1'].forEach(function(d){
        var b = document.createElement('button');
        b.type = 'button';
        b.className = 'azl-arr azl-arr--' + (d === '-1' ? 'l' : 'r');
        b.setAttribute('data-d', d);
        b.setAttribute('aria-label', d === '-1' ? 'Предыдущие новости' : 'Следующие новости');
        b.innerHTML = d === '-1' ? '&#8592;' : '&#8594;';
        lenta.appendChild(b);
        knopki.push(b);
      });
      function poseredineKadra(){
        var ph = cont.querySelector('.t404__imgbox') || cont.querySelector('.t404__img');
        if (!ph) return;
        var r1 = ph.getBoundingClientRect(), r0 = lenta.getBoundingClientRect();
        rec.style.setProperty('--azl-mid', Math.round(r1.top - r0.top + r1.height / 2) + 'px');
      }
      window.addEventListener('resize', poseredineKadra);
      setTimeout(poseredineKadra, 150); setTimeout(poseredineKadra, 900);
      function shag(){
        var c = cont.querySelector('.t404__col');
        if (!c) return 320;
        var g = parseFloat(getComputedStyle(cont).gap) || 24;
        return c.getBoundingClientRect().width + g;
      }
      function osvezhit(){
        var max = cont.scrollWidth - cont.clientWidth - 2;
        var listaetsya = max > 4;
        knopki[0].disabled = !listaetsya || cont.scrollLeft <= 2;
        knopki[1].disabled = !listaetsya || cont.scrollLeft >= max;
        poseredineKadra();
      }
      knopki.forEach(function(b){
        b.addEventListener('click', function(){
          cont.scrollBy({left: shag() * parseInt(b.getAttribute('data-d'),10), behavior:'smooth'});
        });
      });
      cont.addEventListener('scroll', osvezhit, {passive:true});
      window.addEventListener('resize', osvezhit);
      setTimeout(osvezhit, 120); setTimeout(osvezhit, 800);
    })();

    var cols = rec.querySelectorAll('.t404__col');
    [].forEach.call(cols, function(col, i){
      col.style.setProperty('--d', i % 3);

      var dt = col.querySelector('.t404__date');
      if (DATY_RU && dt) {
        var r = poRusski(dt.textContent.trim());
        if (r) dt.textContent = r;
      }
      if (UBRAT_EMOJI) {
        [].forEach.call(col.querySelectorAll('.t404__title,.t404__descr'), function(el){
          el.textContent = el.textContent.replace(EMO,'').replace(/\s{2,}/g,' ').replace(/\s+([!?,.;:])/g,'$1').replace(/[\s!]+$/,'').trim();
        });
      }
    });
    snyatVysoty(rec);

    /* появление + страховка: ни одна карточка не остаётся невидимой */
    function pokazat(el){ el.classList.add('azl-in'); }
    if ('IntersectionObserver' in window) {
      var io = new IntersectionObserver(function(es){
        es.forEach(function(e){ if (e.isIntersecting) { pokazat(e.target); io.unobserve(e.target); } });
      }, {rootMargin:'0px 0px -8% 0px', threshold:.05});
      [].forEach.call(cols, function(c){ io.observe(c); });
      setTimeout(function(){ [].forEach.call(cols, pokazat); }, 2200);
    } else {
      [].forEach.call(cols, pokazat);
    }

    /* Тильда ровняет высоты после загрузки картинок — чистим ещё раз */
    [400, 1200, 2500].forEach(function(ms){ setTimeout(function(){ snyatVysoty(rec); }, ms); });
  }

  function start(){
    var demo = document.getElementById('azl-demo');
    if (demo && document.getElementById('allrecords')) demo.remove();  /* на странице Тильды демо не нужно */

    if (SPRYATAT_KODOVYY && document.getElementById(KODOVYY) && !document.getElementById('azl-hide')) {
      var st = document.createElement('style');
      st.id = 'azl-hide';
      st.textContent = '#' + KODOVYY + '{display:none!important}';
      document.head.appendChild(st);
    }

    [].forEach.call(document.querySelectorAll('.r[data-record-type="404"]'), odet);
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', start);
  else start();
  window.addEventListener('load', start);
})();
</script>
